يمكنك نسخ هذا الكود ولصقه مباشرة في Supabase SQL Editor لإنشاء الجداول الـ 10 الحقيقية بروابطها الأصلية وبدون RLS معقد. 

-- ==============================================================================
-- مشروع هيباتيا ونوب: إعادة بناء قاعدة البيانات بصيغة نظيفة ومباشرة
-- الجداول الـ 10 الحقيقية للمشاريع
-- ==============================================================================

CREATE TABLE IF NOT EXISTS public.projects (
    id TEXT PRIMARY KEY,
    number TEXT NOT NULL,
    name TEXT NOT NULL,
    name_en TEXT,
    category TEXT NOT NULL DEFAULT 'موبايل',
    icon TEXT DEFAULT '📱',
    apk_status TEXT,
    dashboard_status TEXT,
    figma_progress INTEGER DEFAULT 0,
    emails_count INTEGER DEFAULT 0,
    emails_list JSONB DEFAULT '[]'::jsonb,
    next_meeting TEXT,
    meeting_link TEXT,
    apk_link TEXT,
    figma_link TEXT,
    dashboard_link TEXT,
    github_link TEXT,
    drive_link TEXT,
    voice_notes TEXT,
    notes TEXT,
    created_at TIMESTAMPTZ DEFAULT NOW()
);

ALTER TABLE public.projects DISABLE ROW LEVEL SECURITY;
